  Component(preprocess, "Preprocess CLI", "preprocess.py", "Normalizes, deduplicates, sessionizes, splits train/val/test, joins metadata, and builds category vocabulary")
  Component(dataset, "Dataset and Cache Builder", "ingestion/dataset.py", "Builds SessionDataset, InteractionGenerator, SKU vocab, SKU properties, category pools, and session caches")
  Component(svdpq, "SVD-PQ Tokenizer", "ingestion/svdpq.py", "Builds weighted user-SKU matrix, SVD factors, quantized SKU tokens, and cold-start fallback tokens")
